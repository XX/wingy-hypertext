          // тонкая generic-обёртка
          render_button_inner(self_как_данные, self.children.as_ref().map(|c| c as &dyn Renderable), buffer);
      }
  }

  fn render_button_inner(/* негенерик-поля */, children: Option<&dyn Renderable>, buffer: &mut Buffer) {

  }
